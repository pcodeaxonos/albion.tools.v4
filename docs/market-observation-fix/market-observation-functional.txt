const timestamps = new Map();
function parsedTime(value) {
    if (timestamps.has(value)) return timestamps.get(value);
    const at = Date.parse(normalizePriceDate(value));
    const result = { at, iso: Number.isFinite(at) ? new Date(at).toISOString() : null };
    if (timestamps.size >= OBSERVATION_POLICY.timestampCacheLimit) timestamps.clear();
    timestamps.set(value, result); return result;
}

/** Source time is evidence; packet receipt time explicitly is not an exchange timestamp. */
export function normalizeObservation(raw, { now = Date.now(), source = raw.source } = {}) {
    const observedAt = raw.observedAt ?? raw.sourceQuoteAt ?? (source === 'aodp-current' ? null : raw.seenAt ?? raw.at);
    const receivedValue = raw.receivedAt ?? raw.fetchedAt ?? raw.seenAt ?? raw.at;
    const observed = parsedTime(observedAt), receipt = parsedTime(receivedValue);
    const at = observed.at, received = receipt.at;
    const price = Number(raw.price);
    const reasons = [];
    if (!Number.isFinite(price) || price <= 0) reasons.push('invalid-price');
    if (receivedValue !== undefined && receivedValue !== null && !Number.isFinite(received)) reasons.push('invalid-receipt-time');
    if (raw.server !== undefined && raw.server !== null && (!raw.server || !raw.itemId || !raw.city || !['buy', 'sell'].includes(raw.side) || !Number.isInteger(raw.quality) || raw.quality < 1 || raw.quality > 5)) reasons.push('invalid-series');
    if (!Number.isFinite(at) || at > now || (Number.isFinite(received) && at > received)) reasons.push('invalid-source-time');
    if (Number.isFinite(received) && received > now) reasons.push('future-receipt');
    const coverage = raw.coverage ?? (source === 'aodp-current' ? 'reported-top' : 'unknown');
    const independent = source === 'aodp-current' || raw.sourceTimeKnown === true;
    if (!independent) reasons.push('source-time-unknown');
    if (!['reported-top', 'complete'].includes(coverage)) reasons.push('incomplete-book');
    const rejected = reasons.some(reason => reason.startsWith('invalid-') || reason.startsWith('future-'));
    return { server: raw.server ?? null, itemId: raw.itemId ?? null, city: raw.city ?? null, quality: raw.quality ?? null, side: raw.side ?? null,
        seenAt: raw.seenAt ?? null, bucketAt: raw.bucketAt ?? null, fetchedAt: raw.fetchedAt ?? null, sourceQuoteAt: raw.sourceQuoteAt ?? null, quotedAt: raw.quotedAt ?? null, evidence: raw.evidence ?? [],
        observationVersion: OBSERVATION_POLICY.version, source, price,
        observedAt: independent ? observed.iso : null,
        receivedAt: receipt.iso,
        processedAt: raw.processedAt ?? parsedTime(new Date(now).toISOString()).iso, sourceTimeKnown: independent, coverage,
        observationId: raw.observationId ?? `${marketSeriesKey(raw)}|${source}|${observed.iso ?? observedAt}`,
        evidenceId: raw.evidenceId ?? `${marketSeriesKey(raw)}|market|${observed.iso ?? observedAt}`,
        validationStatus: rejected ? 'rejected' : reasons.length ? 'provisional' : 'accepted', reasons,
        at, rawEvidence: raw.rawEvidence ?? raw };
}

export function referenceLevel(count, span, stability) {
    if (!count) return 0;
    let level = 1;
    OBSERVATION_POLICY.levels.forEach(([samples, duration], index) => {
        if (count >= samples && span >= duration && stability >= .8) level = index + 1;
    });
    return level;
}

/** Select by evidence eligibility and exchange time, never by a fixed source-name ranking.
 * A credible pending/quarantined current event is not hidden behind an older accepted price.
 */
export function selectCurrentObservation(observations) {
    const credible = observations.filter(point => point.sourceTimeKnown && ['reported-top', 'complete'].includes(point.coverage)
        && point.validationStatus !== 'rejected');
    const latest = credible.at(-1) ?? observations.at(-1) ?? null;
    const sourceDecisions = [];
    if (latest) for (const point of observations) {
        if (point === latest || point.source === latest.source || point.price === latest.price) continue;
        const reasons = ['source-conflict'];
        if (point.validationStatus === 'rejected') reasons.push('source-invalid');
        else if (!point.sourceTimeKnown || !['reported-top', 'complete'].includes(point.coverage)) reasons.push('source-ineligible');
        else if (point.at < latest.at) reasons.push('source-older');
        sourceDecisions.push({ observationId: point.observationId, evidenceId: point.evidenceId, source: point.source,
            price: point.price, at: point.observedAt, receivedAt: point.receivedAt, selectedObservationId: latest.observationId, reasons });
    }
    return { latest, sourceDecisions };
}

/** Replay admission chronologically. Confirmation changes future references, never its own baseline. */
const referenceCache = new WeakMap();
let cachedPointCount = 0;
export function validateObservationSeries(points, { now = Date.now() } = {}) {
    const cached = referenceCache.get(points);
    if (cached && now >= cached.at && now < cached.expiresAt) return cached.result;
    const policy = OBSERVATION_POLICY;
    const unique = new Map(), conflicts = new Set();
    for (const raw of points || []) {
        const point = normalizeObservation(raw, { now });
        if (!Number.isFinite(point.at) || point.at < now - policy.windowMs || point.at > now) continue;
        const key = point.evidenceId, previous = unique.get(key);
        if (!previous) unique.set(key, point);
        else if (previous.price !== point.price || conflicts.has(key)) {
            conflicts.add(key);
            unique.set(key, { ...previous, validationStatus: 'quarantined', reasons: ['conflicting-event'] });
        } else if (previous.validationStatus !== 'accepted' && point.validationStatus === 'accepted') unique.set(key, point);
    }
    const observations = [...unique.values()].sort((a, b) => a.at - b.at || a.observationId.localeCompare(b.observationId));
    const { latest, sourceDecisions } = selectCurrentObservation(observations);
    let admitted = [], pending = [], transitions = [], anchor = null, anchorMature = false;
    let currentReference = null, currentHistory = [], currentMove = [], bootstrapEstablished = false;
    const metrics = (history, reference) => {
        const spanMs = history.length > 1 ? history.at(-1).at - history[0].at : 0;
        const stability = reference > 0 && history.length ? history.filter(p => Math.abs(p.price-reference)/reference <= policy.flatBand).length/history.length : 0;
        return { spanMs, stability, level: referenceLevel(history.length, spanMs, stability) };
    };
    const admit = candidates => {
        const ids = new Set(admitted.map(p => p.evidenceId));
        for (const candidate of candidates) if (!ids.has(candidate.evidenceId)) { admitted.push(candidate); ids.add(candidate.evidenceId); }
        admitted.sort((a,b)=>a.at-b.at);
    };
    for (const point of observations) {
        if (point.sourceTimeKnown && ['reported-top','complete'].includes(point.coverage) && point.validationStatus !== 'rejected') {
            const poolReference = median(admitted.map(p=>p.price));
            const reference = pending.length && anchorMature ? anchor : poolReference;
            // Snapshot before this event: even retroactively accepted witnesses cannot validate this event's reference.
            const history = [...admitted];
            let move = [point];
            if (point.validationStatus === 'accepted') {
                const comparison = pending.length ? anchor : reference;
                const deviation = comparison > 0 ? Math.abs(point.price-comparison)/comparison : 0;
                const departure = deviation > policy.flatBand && Math.abs(point.price-comparison) >= policy.minDelta;
                if (!departure) {
                    admit([point]); pending=[]; anchor=null; anchorMature=false;
                } else {
                    if (pending.length && Math.abs(point.price-median(pending.map(p=>p.price)))/point.price > policy.flatBand) {
                        pending=[]; anchor=null; anchorMature=false;
                    }
                    if (!pending.length) { anchor=reference; anchorMature=metrics(history,reference).level>=3; }
                    pending.push(point); move=[...pending];
                    const departureSize = anchor>0 ? Math.abs(point.price-anchor)/anchor : 0;
                    const required = departureSize > policy.extreme ? policy.extremeConfirmations : departureSize >= policy.highSuspicion ? policy.highSuspicionConfirmations : departureSize >= policy.suspicion ? policy.suspiciousConfirmations : policy.confirmations;
                    const span = pending.at(-1).at-pending[0].at;
                    const verified = pending.length>=required && span >= (required-1)*3600000;
                    point.validationStatus=verified?'accepted':departureSize>=policy.suspicion?'quarantined':'provisional';
                    point.reasons=verified?['movement-confirmed']:[departureSize>policy.extreme?'extreme-price':departureSize>=policy.suspicion?'suspicious-movement':'awaiting-movement'];
                    point.evidence=pending.map(p=>p.observationId);
                    if (verified) {
                        for (const candidate of pending) { candidate.validationStatus='accepted'; candidate.reasons=['movement-confirmed']; candidate.evidence=point.evidence; }
                        admit(pending);
                    }
                    if (verified && pending.length>=policy.regimeSamples && span>=policy.regimeSpanMs) {
                        transitions.push({ type:'regime', at:point.observedAt, from:anchor, to:median(pending.map(p=>p.price)), evidence:point.evidence });
                        admitted=[...pending]; pending=[]; anchor=null; anchorMature=false;
                    }
                }
            }
            const pool = median(admitted.map(p=>p.price));
            if (!bootstrapEstablished && metrics(admitted,pool).level>=3) {
                bootstrapEstablished=true;
                transitions.push({ type:'bootstrap', at:point.observedAt, from:observations.find(p=>p.validationStatus==='accepted')?.price??null,
                    to:pool, evidence:admitted.map(p=>p.observationId) });
            }
            if (point === latest) { currentReference=reference; currentHistory=history; currentMove=move; }
        } else if (point === latest) { currentReference=median(admitted.map(p=>p.price)); currentHistory=[...admitted]; currentMove=[point]; }
    }
    const {spanMs,stability,level}=metrics(currentHistory,currentReference);
    const result={observations,latest,sourceDecisions,reference:currentReference>0?currentReference:null,samples:currentHistory.length,
        spanMs,stability,level,confidence:level/4,provisional:level<3,move:currentMove,transitions,
        referenceEvidence:currentHistory.map(p=>p.observationId),
        executionConfidence:latest?.validationStatus==='accepted' && now-latest.at<=PRICE_STALE_MS?1:0};
    const boundaries=observations.map(point=>point.at+policy.windowMs).filter(at=>at>now);
    if (cachedPointCount+observations.length<=policy.maxCachedPoints || cached) {
        if(!cached)cachedPointCount+=observations.length;
        referenceCache.set(points,{at:now,expiresAt:Math.min(now+3600000,...boundaries,latest?latest.at+PRICE_STALE_MS:Infinity),result});
    }
    return result;
}
